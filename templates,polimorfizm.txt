template <typename T>class Process {
private:
	T* array;
	int counter = 0;
public:
	Process() {
		this->array = new T[20];
	}


	void Add(T num) {
		this->array[counter] = num;
		counter++;
	}

	void Show() {
		for (int i = 0; i < counter; i++)
		{
			cout << this->array[i] << " ";
		}
		cout << endl;
	}

	void Max() {
		T max = this->array[counter - 1];
		for (int i = 0; i < counter; i++)
		{
			if (max < this->array[i]) {
				max = this->array[i];
			}
		}
		cout << max << endl;

		//Second method

		//Bubble_sort();
		//cout << this->array[counter - 1] <<endl;
	}

	void Min() {
		T min = this->array[counter - 1];
		for (int i = 0; i < counter; i++)
		{
			if (min > this->array[i]) {
				min = this->array[i];
			}
		}
		cout << min << endl;

		//Second method

		//Bubble_sort();
		//cout << this->array[0] <<endl;
	}

	void Bubble_sort() {
		for (int step = 0; step < counter - 1; step++)
		{

			bool flag = true;

			for (int i = 0; i < (counter - step - 1); i++)
			{
				if (this->array[i] > this->array[i + 1]) {
					int temp = this->array[i];
					this->array[i] = this->array[i + 1];
					this->array[i + 1] = temp;

					flag = false;
				}
			}

			if (flag) {
				break;
			}
		}
	}

	void Search(T num) {
		Bubble_sort();
		int result = Binary_search(num, 0, counter - 1);
		if (result == -1) {
			cout << "Takoqo cisla net" << endl;
		}
		else {
			cout << "Vashe cislo " << "{" << this->array[result] << "}" << " na pozicii -> " << "[" << result << "]";
		}
	}

	int Binary_search(T cislo, int low, int high) {
		if (high >= low) {
			int mid = low + (high - low) / 2;

			if (cislo == this->array[mid]) {
				return mid;
			}

			if (cislo > this->array[mid]) {
				return Binary_search(cislo, mid + 1, high);
			}

			return Binary_search(cislo, low, mid - 1);
		}

		return -1;
	}

	void Change(T oldNum, T newNum) {
		Bubble_sort();
		int find = Binary_search(oldNum, 0, counter - 1);
		if (find == -1) {
			cout << "Takoqo cisla net" << endl;
		}
		else {
			this->array[find] = newNum;
			cout << "Zamena proizosla!" << endl;
		}
	}


	~Process() { delete[]array; }

};

class Student {
private:
	char* name;
	int grade;
public:
	Student(const char* name, int grade) {
		this->name = new char[strlen(name) + 1];

		strcpy_s(this->name, strlen(name) + 1, name);
		this->grade = grade;
	}

	~Student() {
		delete[]name;
	}

	void Info() {
		cout << "Student: " << this->name << ";" << " Grade: " << this->grade << endl;

	}

	Student(Student&& other) {
		this->name = other.name;
		this->grade = other.grade;

		other.name = nullptr;
		other.grade = 0;
	}

	int GetGrade() {
		return this->grade;
	}

	friend void Increase(Student& student);
};

class Person {
private:
	char* name;
	char* surname;
	char* father_name;
	int age;
public:
	Person(const char* name, const char* surname, const char* father_name, int age) {
		this->name = new char[strlen(name) + 1];
		this->surname = new char[strlen(surname) + 1];
		this->father_name = new char[strlen(father_name) + 1];

		strcpy_s(this->name, strlen(name) + 1, name);
		strcpy_s(this->surname, strlen(surname) + 1, surname);
		strcpy_s(this->father_name, strlen(father_name) + 1, father_name);
		this->age = age;
	}

	Person(Person&& other) {
		this->name = other.name;
		this->surname = other.surname;
		this->father_name = other.father_name;
		this->age = other.age;

		other.name = nullptr;
		other.surname = nullptr;
		other.father_name = nullptr;
		other.age = 0;
	}

	Person() {
		this->name = nullptr;
		this->surname = nullptr;
		this->father_name = nullptr;
		this->age = 0;
	}

	Person& operator=(const Person& other) {
		delete[] name;
		delete[] surname;
		delete[] father_name;

		this->name = new char[strlen(other.name) + 1];
		this->surname = new char[strlen(other.surname) + 1];
		this->father_name = new char[strlen(other.father_name) + 1];

		strcpy_s(this->name, strlen(other.name) + 1, other.name);
		strcpy_s(this->surname, strlen(other.surname) + 1, other.surname);
		strcpy_s(this->father_name, strlen(other.father_name) + 1, other.father_name);
		this->age = other.age;
		return *this;
	}

	~Person() {
		delete[] name;
		delete[] surname;
		delete[] father_name;
	}

	void show_Person() {
		cout << this->name << " " << this->surname << " " << this->father_name << " " << this->age << endl;
	}

};

class Flat {
private:
	Person* persons;
	int people;
	int amount_room;
	float size_of_flat;
	int flat_counter;
public:
	Flat(int people, int amount_room, float size_of_flat) {
		this->persons = new Person[people];
		this->people = people;
		this->amount_room = amount_room;
		this->size_of_flat = size_of_flat;
		this->flat_counter = 0;
	}

	Flat(Flat&& other) {
		this->persons = other.persons;
		this->people = other.people;
		this->amount_room = other.amount_room;
		this->size_of_flat = other.size_of_flat;
		this->flat_counter = this->flat_counter;



		other.persons = nullptr;
		other.people = NULL;
		other.amount_room = NULL;
		other.size_of_flat = NULL;
		other.flat_counter = NULL;
	}


	Flat() {
		this->persons = nullptr;
		this->people = 0;
		this->amount_room = 0;
		this->size_of_flat = 0;
		this->flat_counter = 0;
	}

	Flat& operator=(const Flat& other) {
		delete[] persons;

		this->people = other.people;
		this->amount_room = other.amount_room;
		this->size_of_flat = other.size_of_flat;
		this->flat_counter = other.flat_counter;

		persons = new Person[people];
		for (int i = 0; i < people; ++i) {
			this->persons[i] = other.persons[i];
		}

		return *this;
	}

	void Add_person(const Person& other) {
		if (this->people > this->flat_counter) {

			persons[flat_counter] = other;
			flat_counter++;
		}
		else {
			cout << "Kvartira ponlnaya!";
		}
	}

	void show_Falt() {
		if (this->people > 0) {
			cout << "Kolicestvo komnat v kvartire: " << this->amount_room << endl;
			cout << "Kvadratura kvartiti: " << this->size_of_flat << endl;
			cout << "Kolicestvo jiteley - > " << this->people << endl;
			cout << endl;
			for (int i = 0; i < this->people; i++)
			{
				persons[i].show_Person();

			}
		}
		else {
			cout << "Kvartira pusta!";
		}

	}

	~Flat() {
		delete[]persons;
	}
};


class House {
private:
	Flat* flats;
	int amount_of_flat;
	int amount_floar;
	int amount_entrance;
	int house_counter;
public:
	House(int amount_of_flat, int amount_floar, int amount_entrance) {
		this->flats = new Flat[amount_of_flat];
		this->amount_of_flat = amount_of_flat;
		this->amount_floar = amount_floar;
		this->amount_entrance = amount_entrance;
		this->house_counter = 0;
	}

	House(House&& other) {
		this->flats = other.flats;
		this->amount_of_flat = other.amount_of_flat;
		this->amount_floar = other.amount_floar;
		this->amount_entrance = other.amount_entrance;
		this->amount_of_flat = other.house_counter;



		other.flats = nullptr;
		other.amount_of_flat = 0;
		other.amount_floar = 0;
		other.amount_entrance = 0;
		other.house_counter = 0;
	}

	House() {
		this->flats = nullptr;
		this->amount_of_flat = 0;
		this->amount_floar = 0;
		this->amount_entrance = 0;
		this->house_counter = 0;
	}

	void Add_flats(const Flat& other) {
		if (this->amount_of_flat > this->house_counter) {

			flats[house_counter] = other;
			house_counter++;
		}
		else {
			cout << "Dom ponlniy!";
		}
	}

	void show_House() {
		if (this->amount_of_flat > 0) {
			cout << "Kolicestvo etajey: " << this->amount_floar << endl;
			cout << "Kolicestvo podezdov: " << this->amount_entrance << endl;
			cout << "Kolicestvo kvartir v dome - > " << this->amount_of_flat << endl;
			cout << endl;
			for (int i = 0; i < this->amount_of_flat; i++)
			{
				flats[i].show_Falt();
				cout << endl;
			}
		}
		else {
			cout << "takoqo doma net!";
		}

	}

	~House() {
		delete[]flats;
	}
};
