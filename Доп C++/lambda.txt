#include <iostream>

int main() {
    // Lambda function that takes two integers and returns their sum
    auto add = [](int a, int b) { return a + b; };

    // Using the lambda function
    int result = add(5, 3);
    std::cout << "Result: " << result << std::endl;

    return 0;
}

#include <iostream>

int main() {
    int x = 10;
    int y = 20;

    // Lambda function capturing x and y by value
    auto add = [x, y]() { return x + y; };

    // Using the lambda function
    std::cout << "Sum: " << add() << std::endl;

    return 0;
}

#include <iostream>

int main() {
    // Lambda function taking two parameters and returning their product
    auto multiply = [](int a, int b) { return a * b; };

    // Using the lambda function
    std::cout << "Product: " << multiply(3, 4) << std::endl;

    return 0;
}

#include <iostream>

int main() {
    // Lambda function without parameters
    auto hello = []() { std::cout << "Hello, World!" << std::endl; };

    // Using the lambda function
    hello(); // Prints: Hello, World!

    return 0;
}
