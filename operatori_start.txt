// Task 1
class Fraction {
private:
	int numerator;
	int denominator;
	double doub;
public:
	Fraction(int numerator, int denominator) : numerator(numerator), denominator(denominator) {
		this->doub = static_cast<double>(numerator) / denominator;
	}

	Fraction(): numerator(0), denominator(0), doub(0){}


	Fraction& operator +(const Fraction& other) {
		Fraction temp;
		temp.doub= this->doub + other.doub;
		return temp;
	}

	Fraction& operator -(const Fraction& other) {
		Fraction temp;
		temp.doub = this->doub - other.doub;
		return temp;
	}

	Fraction& operator *(const Fraction& other) {
		Fraction temp;
		temp.doub = this->doub * other.doub;
		return temp;
	}

	Fraction& operator /(const Fraction& other) {
		Fraction temp;
		temp.doub = this->doub / other.doub;
		return temp;
	}

	void Show() {
		cout << "Number: " << this->doub << endl;
	}
};

class Data {
private:
	unsigned short year;
	unsigned short month;
	unsigned short day;
public:
	Data(unsigned short year, unsigned short month, unsigned short day) : year(year), month(month), day(day) {

	}

	Data() : year(0), month(0), day(0) {}

	void Check_data() {
		if (this->month == 12 && this->day > 31) {
			this->day = 1;
			this->month = 1;
			this->year++;
		}
		else if (this->month == 1 && this->day < 1) {
			this->day = 31;
			this->month = 12;
			this->year--;
		}
		else if (this->day > 31) {
			this->day = 1;
			this->month++;
		}
		else if (this->day < 1) {
			this->day = 31;
			this->month--;
		}
	}

	void Next_Day() {
		this->day++;
		Check_data();
	}

	Data& operator++ ()
	{
		this->day++;
		Check_data();
		return *this;
	}
	Data& operator-- ()
	{
		this->day--;
		Check_data();
		return *this;
	}

	Data operator++ (int)
	{
		Data copy{ *this };
		++(*this);
		copy.Check_data();
		return copy;
	}
	Data operator-- (int)
	{
		Data copy{ *this };
		--(*this);
		copy.Check_data();
		return copy;
	}

	bool operator != (const Data& other) {
		if (this->day != other.day || this->month != other.month || this->year != other.year) {
			return true;
		}
		else {
			return false;
		}
	}

	bool operator == (const Data& other) {
		if (this->day == other.day && this->month == other.month && this->year == other.year) {
			return true;
		}
		else {
			return false;
		}
	}

	bool operator > (const Data& other) {

		if (this->year > other.year) {
			return true;
		}
		else if (this->year == other.year) {
			if (this->month > other.month) {
				return true;
			}
			else if (this->month == other.month) {
				if (this->day > other.day) {
					return true;
				}
				else {
					return false;
				}
			}
		}
		else {
			return false;
		}
	}

	bool operator < (const Data& other) {

		if (this->year < other.year) {
			return true;
		}
		else if (this->year == other.year) {
			if (this->month < other.month) {
				return true;
			}
			else if (this->month == other.month) {
				if (this->day < other.day) {
					return true;
				}
				else {
					return false;
				}
			}
		}
		else {
			return false;
		}
	}

	friend istream& operator>>(istream& stream, Data& other) {
		return stream >> other.year >> other.month >> other.day;
	}

	friend ostream& operator<<(ostream& stream, Data& other) {

		if (other.month >= 10 && other.month <= 12 && other.day <= 31 && other.day > 0) {
			if (other.day < 10) {
				return stream << "Data: " << other.year << "-" << other.month << "-" << "0" << other.day << endl;
			}
			else {
				return stream << "Data: " << other.year << "-" << other.month << "-" << other.day << endl;
			}
		}
		else if (other.month < 10 && other.month > 0 && other.day <= 31 && other.day > 0) {
			if (other.day < 10 && other.day > 0) {
				return stream << "Data: " << other.year << "-" << "0" << other.month << "-" << "0" << other.day << endl;
			}
			else {
				return stream << "Data: " << other.year << "-" << "0" << other.month << "-" << other.day << endl;
			}
		}
		else {
			return stream << "Nepravilniy vvod!" << endl;
		}
	}

	Data& operator = (const Data& other)
	{
		this->year = other.year;
		this->month = other.month;
		this->day = other.day;
		return *this;
	}

};
