#include <iostream>

using namespace std;

class MyStr {
private:
	char* str;
public:
	MyStr(const char* stroka) {
		this->str = new char[strlen(stroka) + 1];
		strcpy_s(this->str, strlen(stroka) + 1, stroka);
	}
	
    void Append(const char* stroka) {
        int oldSize = strlen(this->str);
        int newSize = strlen(stroka);
        char* newStorka = new char[oldSize + newSize + 1];

      
        for (int i = 0; i < oldSize; i++) {
            newStorka[i] = this->str[i];
        }

        for (int i = 0; i < newSize; i++) {
            newStorka[i + oldSize] = stroka[i];
        }

        newStorka[oldSize + newSize] = '\0';
        delete[] this->str;
        this->str = newStorka; 
    }

    void Clear() {
        delete[]this->str;
        this->str = nullptr;
    }

    bool Empty() {
        return strlen(this->str) == 0;
    }

	void Ltrim() {
		int left = 0;
		for (int i = 0; i < strlen(this->str); i++)
		{
			if (this->str[i] == char(32)) {
				left++;
			}
			else {
				break;
			}
		}

		int newSize = strlen(this->str) - left;
		char *newStroka = new char[newSize + 1];
		newStroka[newSize] = '\0';

		for (int i = 0; i < newSize; i++)
		{
			newStroka[i] = this->str[i + left];
		}
		delete[] this->str;
		this->str = newStroka;
	}

	void Rtrim() {
		int right = 0;
		for (int i = strlen(this->str) - 1; i > 0; i--)
		{
			if (this->str[i] == char(32)) {
				right++;
			}
			else {
				break;
			}
		}
		int newSize = strlen(this->str) - right;
		char* newStroka = new char[newSize + 1];
		newStroka[newSize] = '\0';

		for (int i = 0; i < newSize; i++) {
			newStroka[i] = this->str[i];
		}
		delete[] this->str;
		this->str = newStroka;
	}

	void Trim() {
		int counterAll = 0;

		for (int i = 0; i < strlen(this->str); i++)
		{
			if (this->str[i] == char(32)) {
				counterAll++;
			}
			else {
				continue;
			}
		}



		int newSize = strlen(this->str) - counterAll;

		char* newStroka = new char[newSize + 1];
		newStroka[newSize] = '\0';

		int counter = 0;

		for (int i = 0; i < strlen(this->str); i++)
		{
			if (this->str[i] != char(32)) {
				newStroka[counter] = this->str[i];
				counter++;
			}
		}

		delete[] this->str;
		this->str = newStroka;

	}

	void Upper() {
		for (int i = 0; i < strlen(this->str); i++)
		{
			if (96 < static_cast<int>(this->str[i]) && static_cast<int>(this->str[i]) < 123) {
				this->str[i] = char(static_cast<int>(this->str[i]) - 32);
			}
		}
		cout << this->str;
	}

	void Lower() {
		for (int i = 0; i < strlen(this->str); i++)
		{
			if (64 < static_cast<int>(this->str[i]) && static_cast<int>(this->str[i]) < 91) {
				this->str[i] = char(static_cast<int>(this->str[i]) + 32);
			}
		}
		cout << this->str;
	}

	void Reverse() {
		for (int i = strlen(this->str); i >= 0; i--)
		{
			cout << this->str[i];
		}
	}

	const char* GetStr() {
		return this->str;
	}

    ~MyStr() {
        delete[]this->str;
    }
};


int main() {
	MyStr name("Hello");
	name.Reverse();

	cout << name.GetStr() << endl;

	name.Append("asda");
	cout << name.GetStr()<<endl;
    name.Append("asda");
    cout << name.GetStr();

	return 0;
}
