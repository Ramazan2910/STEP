const char* Concatenation(const char* old_one, const char* new_one) {
	int oldSize = strlen(old_one);
	int newSize = strlen(new_one);
	char* new_stroka = new char[oldSize + newSize + 1];


	for (int i = 0; i < oldSize; i++) {
		new_stroka[i] = old_one[i];
	}

	for (int i = 0; i < newSize; i++) {
		new_stroka[i + oldSize] = new_one[i];
	}

	new_stroka[oldSize + newSize] = '\0';

	return new_stroka;
}

const char* Encryption(const char* text) {

	char* encrypted_text = new char[strlen(text) + 1];

	for (int i = 0; i < strlen(text); i++) {
		encrypted_text[i] = text[i] + 3;
	}

	encrypted_text[strlen(text)] = '\0';

	return encrypted_text;
}

const char* Decryption(const char* text) {
	char* dencrypted_text = new char[strlen(text) + 1];

	for (int i = 0; i < strlen(text); i++) {
		dencrypted_text[i] = text[i] - 3;
	}

	dencrypted_text[strlen(text)] = '\0';

	return dencrypted_text;
}

	bool CheckLoginPassword(const char* filename, const char* login, const char* password) override {
		int index = 0;
		Student* users = Read(filename, index);
		for (int i = 0; i < index; i++)
		{
			if (strcmp(users[i].login, login) == 0 && strcmp(users[i].password, password) == 0) {
				return true;
				delete[]users;
			}
		}
		return false;
		delete[]users;
	}

	bool Check_for_correctness_login(const char* filename, const char* login) {
		// Leght of strnigs 

		if ((strlen(login) < 7 || strlen(login) > 20)) {
			return false;
		}
		else {
			bool login_flag = false;
			int alpha_count = 0;
			int digit_count = 0;
			int special_simbol_counter = 0;

			for (int i = 0; i < strlen(login); i++)
			{
				if ((char(64) < login[i] && login[i] < char(64)) || (char(96) < login[i] && login[i] < char(123))) {
					alpha_count++;
				}
				else if (0 <= login[i] - '0' && login[i] - '0' < 10) {
					digit_count++;
				}
				else if (login[i] == '@' || login[i] == '%' || login[i] == '$' || login[i] == '~' || login[i] == '&' || login[i] == '*' || login[i] == '#') {
					special_simbol_counter++;
				}
			}
			if (alpha_count > 0 && digit_count > 0 && special_simbol_counter > 0) {
				login_flag = true;
			}

			return login_flag;
		}
	}

	bool Check_for_correctness_password(const char* filename, const char* password) {

		if ((strlen(password) < 7 || strlen(password) > 20)) {
			return false;
		}
		else {
			bool password_flag = false;
			bool unique_pass;
			int alpha_count = 0;
			int digit_count = 0;
			int special_simbol_counter = 0;

			for (int i = 0; i < strlen(password); i++)
			{
				if ((char(64) < password[i] && password[i] < char(64)) || (char(96) < password[i] && password[i] < char(123))) {
					alpha_count++;
				}
				else if (0 <= password[i] - '0' && password[i] - '0' < 10) {
					digit_count++;
				}
				else if (password[i] == '@' || password[i] == '%' || password[i] == '$' || password[i] == '~' || password[i] == '&' || password[i] == '*' || password[i] == '#') {
					special_simbol_counter++;
				}
			}

			if (alpha_count > 0 && digit_count > 0 && special_simbol_counter > 0) {
				password_flag = true;
			}

			unique_pass = Unique_password(filename, password);


			if (password_flag && unique_pass) {
				return true;
			}
			else {
				return false;
			}
		}



	}

template<typename T>
class Node {
public:
	T data;
	Node* next;

	Node(const T& newData) : data(newData), next(nullptr) {}
};


template<typename T>
class List {
private:
	Node<T>* head;

public:
	List() : head(nullptr) {}

	~List() { clear(); }

	void Add(const T& data) {
		if (!head) {
			head = new Node<T>(data);
		}
		else {
			Node<T>* current = head;
			while (current->next) {
				current = current->next;
			}
			current->next = new Node<T>(data);
		}
	}

	void ReadTestFromFileToList(const char* path) {
		Test test;
		Test* test2 = test.ReadTestFromFile(path);


		if (!head) {
			head = new Node<T>(*test2);
		}
		else {
			Node<T>* current = head;
			while (current->next) {
				current = current->next;
			}
			current->next = new Node<T>(*test2);
		}
	}


	void WriteTestToFile(const char* topic, const char* path) {
		Node<T>* current = head;
		while (current) {
			if (strcmp(current->data.GetTopicOfTest(), topic) == 0) {
				current->data.WriteTestToFile(path);
			}
			current = current->next;
		}
	}

	void Edit(const char* filepath, const char* data, int edit_counter) {
		Node<T>* current = head;
		while (current) {
			if (strcmp(current->data.GetTopicOfTest(), data) == 0) {
				current->data.Edit(filepath, edit_counter);
			}
			current = current->next;
		}
	}

	void TakeTheTest(const char* filename, const char* topic_of_test, const char* category, const char* login, const char* password) {
		Node<T>* current = head;
		while (current) {
			if (strcmp(current->data.GetTopicOfTest(), topic_of_test) == 0) {
				current->data.TakeTheTest(filename, category, login, password);
			}
			current = current->next;
		}
	}

	void ShowList() const {
		Node<T>* current = head;
		while (current) {
			current->data.Show();
			current = current->next;
		}
	}

	void ShowStatistica(const char* category, const char* test, const char* password, int choice) const {
		Node<T>* current = head;
		if (choice == 1) {
			while (current) {
				if (strcmp(current->data.GetCategory(), category) == 0) {
					current->data.Show();
				}
				current = current->next;
			}
		}
		else if (choice == 2) {
			while (current) {
				if (strcmp(current->data.GetCategory(), category) == 0 && strcmp(current->data.GetTest(), test) == 0) {
					current->data.Show();
				}
				current = current->next;
			}
		}
		else if (choice == 3) {
			while (current) {
				if (strcmp(current->data.GetPassword(), password) == 0) {
					current->data.Show();
				}
				current = current->next;
			}
		}
	}

	void Remove(const char* data) {
		Node<T>* current = head;
		Node<T>* prev = nullptr;
		while (current) {
			if (strcmp(current->data.GetTopicOfTest(), data) == 0) {
				if (prev) {
					prev->next = current->next;
				}
				else {
					head = current->next;
				}
				delete current;
				return;
			}
			prev = current;
			current = current->next;
		}
	}

	void clear() {
		while (head) {
			Node<T>* temp = head;
			head = head->next;
			delete temp;
		}
	}
};
