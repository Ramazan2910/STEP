struct Step_store {
	char brand[20];
	char model[20];
	int amount;
	int price;
};

Step_store* Read(const char* filename, int& edit_index) {
	Step_store* goods = new Step_store[20];
	FILE* file = nullptr;
	fopen_s(&file, filename, "r");


	char* bfr = new char[100];
	if (file != nullptr) {
		while (fgets(bfr, 100, file) != nullptr) {
			int i = 0;

			for (; bfr[i] != ' '; i++)
			{
				goods[edit_index].brand[i] = bfr[i];
			}
			goods[edit_index].brand[i] = '\0';
			i++;



			int j = 0;
			for (; bfr[i] != ' '; i++, j++)
			{
				goods[edit_index].model[j] = bfr[i];
			}
			goods[edit_index].model[j] = '\0';
			i++;




			int k = 0;
			char* num1 = new char[5];
			for (; bfr[i] != ' '; i++, k++)
			{
				num1[k] = bfr[i];
			}
			num1[k] = '\0';
			goods[edit_index].amount = atoi(num1);
			i++;




			int c = 0;
			char* num2 = new char[8];
			for (; bfr[i] != '\n'; i++, c++)
			{
				num2[c] = bfr[i];
			}
			num2[c] = '\0';
			goods[edit_index].price = atoi(num2);
			i++;


			edit_index++;

		}
		fclose(file);
		delete[]bfr;
	}
	return goods;
}

int Binary_search_brand(Step_store mass[], const char* brand, const char* model, int low, int high) {

	if (high >= low) {
		int mid = low + (high - low) / 2;

		if (strcmp(brand, mass[mid].brand) == 0 && strcmp(model, mass[mid].model) == 0) {
			return mid;
		}
		if (strcmp(brand, mass[mid].brand) == 1 && strcmp(model, mass[mid].model) == 1) {
			return Binary_search_brand(mass, brand, model, mid + 1, high);
		}

		return Binary_search_brand(mass, brand, model, low, mid - 1);
	}

	return -1;
}

void Add(Step_store element, const char* filename) {
	FILE* file = nullptr;
	fopen_s(&file, filename, "a");

	if (file != nullptr) {
		fputs(element.brand, file);
		fputs(" ", file);
		fputs(element.model, file);
		fputs(" ", file);
		char* bfr = new char[10];
		_itoa_s(element.amount, bfr, 10, 10);
		fputs(bfr, file);
		fputs(" ", file);
		char* bfr1 = new char[10];
		_itoa_s(element.price, bfr1, 10, 10);
		fputs(bfr1, file);
		fputs("\n", file);
		fclose(file);
	}

}
