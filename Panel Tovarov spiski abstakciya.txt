#include <iostream>
#include <string>

using namespace std;

class Display {
public:
    virtual void display() = 0;
    virtual ~Display() {}
};

class Item {
protected:
    char* brand;
    char* model;
    int count;
    float price;
public:
    Item() {}

    Item(const char* brand, const char* model, int count, float price) :count(count), price(price) {
        this->brand = new char[strlen(brand) + 1];
        this->model = new char[strlen(model) + 1];

        strcpy_s(this->brand, strlen(brand) + 1, brand);
        strcpy_s(this->model, strlen(model) + 1, model);
    }

    virtual void Show() {
        cout << "Brand: " << this->brand << "; " << "Model: " << this->model << "; " << "Count: " << this->count << "; " << "Price: " << this->price << endl;
    }

    ~Item() {}
};

class Phone : public Item, public Display{
private:
    char* seriyniyNomer;
public:
    Phone() {};

    Phone(const char* brand, const char* model, int count, float price, const char* seriyniyNomer): Item(brand, model, count, price) {
        this->seriyniyNomer = new char[strlen(seriyniyNomer) + 1];

        strcpy_s(this->seriyniyNomer, strlen(seriyniyNomer) + 1, seriyniyNomer);
    }

    void display() override{
        cout << "1222X1880" << endl;
    }

    void Show() override {
        cout << "Brand: " << this->brand << "; " << "Model: " << this->model << "; " << "Count: " << this->count << "; " << "Price: " << this->price << "; " << "Seria: " << this->seriyniyNomer << endl;
    }

    const char* GetBrand() {
        return this->brand;
    }

    void ChangeData(const char* old) {
        delete[]this->brand;
        this->brand = new char[strlen(brand) + 1];
        strcpy_s(this->brand, strlen(old) + 1, old);
    }

    ~Phone(){}
};

class Tools {
public:
    virtual void Add() = 0;
    virtual void Edit(const char* oldone, const char* newone) = 0;
    virtual void Remove(const char* del) = 0;
    virtual void Show() = 0;
    virtual ~Tools() {}
};

class Node {
public:
    Phone data;
    Node* next;

    Node(Phone data) :data(data), next(nullptr) {}
};

class List: public Tools{
private:
    Node* head;
public:
    List():head(nullptr){}

    void Add() override {
        char brand[50];
        char model[50];
        char seria[50];
        int count;
        float price;

        cout << "Enter brand: ";
        cin >> brand;
        cout << "Enter model: ";
        cin >> model;
        cout << "Enter count: ";
        cin >> count;
        cout << "Enter price: ";
        cin >> price;
        cout << "Enter seria: ";
        cin >> seria;

        Phone* newPhone = new Phone(brand, model, count, price, seria);
        Node* newNode = new Node(*newPhone);
        newNode->next = head;
        head = newNode;
    }

    void Remove(const char* del) override {
        Node* current = head;
        Node* prev = nullptr;
        while (current) {
            if (strcmp(current->data.GetBrand(), del) == 0) {
                if (prev) {
                    prev->next = current->next;
                }
                else {
                    head = current->next;
                }
                delete current;
                return;
            }
            prev = current;
            current = current->next;
        }
    }

    void Edit(const char* oldone, const char* newone) override {
        Node* current = head;
        while (current) {
            if (strcmp(current->data.GetBrand(), oldone) == 0) { // Сравнение строк с помощью strcmp
                current->data.ChangeData(newone);
                return;
            }
            current = current->next;
        }
    }

    void Show() override {
        Node* temp = head;
        while (temp) {
            temp->data.Show();
            temp = temp->next;
        }
    }

    ~List() {
        while (head) {
            Node* temp = head;
            head = head->next;
            delete temp;
        }
    }
};

int main() {
    
    List list;
    bool flag = true;

    while (flag) {
        int choise;
        cout << "\tMenu\n1.Add\n2.Remove\n3.Edit\n4.Show\n5.Exit\n";
        cout << "Your choise: ";
        cin >> choise;

        switch (choise)
        {
        case 1:
            list.Add();
            break;
        case 2:
            char brand[50];
            cout << "Enter delete: ";
            cin >> brand;
            list.Remove(brand);
            break;
        case 3:
            char oldone[50];
            char newone[50];
            cout << "Enter which you want change: ";
            cin >> oldone;
            cout << "Enter to which you want change: ";
            cin >> newone;
            list.Edit(oldone, newone);
            break;
        case 4:
            list.Show();
            break;
        case 5:
            flag = false;
            break;
        default:
            break;
        }
    }

    return 0;
}
