
#include <iostream>
#include <string>

using namespace std;

// Task 1


void Change(int* fmassiv, int* fmassiv2, int frange) {
	int fcounter = 0;
	for (int i = 0; i < frange; i++)
	{
		if (*(fmassiv + i) % 2 == 0) {
			*(fmassiv2 + fcounter) = *(fmassiv + i);
			fcounter++;
		}
	}

	cout << "Vot cetniye elemnti - > ";

	for (int i = 0; i < fcounter; i++)
	{
		cout << *(fmassiv2 + i) << " ";
	}
}

void Delete(int* fmassiv) {
	delete[]fmassiv;
}

int main() {

	// Task 1


	int counter = 0;
	int range;
	cout << "Vvedite razmer massiva: ";
	cin >> range;
	int* massiv = new int[range] {};

	for (int i = 0; i < range; i++)
	{
		cout << "Vvedite element massiva: ";
		cin >> *(massiv + i);
		if (*(massiv + i) % 2 == 0) {
			counter++;
		}
	}

	if (counter > 0) {
		int* massiv2 = new int[counter] {};
		Change(massiv, massiv2, range);
		Delete(massiv);
	}
	else {
		cout << "Net cetnix cisel -> ";

		for (int i = 0; i < range; i++)
		{
			cout << *(massiv + i) << " ";
		}
	}

	// Task 2


	int counter = 0;
	string* stroka = new string;
	cout << "Vvedite stroku: ";
	getline(cin, *stroka);



	for (char& i : *stroka) {
		if (i == ' ') {
			counter++;
		}
	}


	cout << "Vot stolko tut slov -> " << ++counter;


	// Task 3

	bool flag = true;
	char stroka[] = { 'h','e','e','h' };
	int size = (sizeof(stroka) / sizeof(stroka[0])) - 1;
	int reverseCounter = size;


	for (int i = 0; i <= size; i++)
	{
		if (stroka[i] == stroka[reverseCounter]) {
			reverseCounter--;
		}
		else {
			cout << "Eto ne palindrom!";
			flag = false;
			break;
		}
	}

	if (flag) {
		cout << "Eto palindrom!";
	}


	// Task 4

	int vowelsCount = 0;
	int consonantsCount = 0;

	char vowels[] = { 'A', 'a', 'E', 'e', 'U', 'u', 'I', 'i', 'O', 'o', 'Y', 'y' };
	char consonants[] = { 'B','b','C','c','D','d','F','f','G','g','H','h','J','j','K','k','L','l','M','m','N','n','P','p','Q','q','R','r','T','t','V','v','W','w','X','x','Z','z' };

	string* stroka = new string;
	cout << "Vvedite stroku: ";
	getline(cin, *stroka);

	for (char& i : *stroka)
	{
		for (char& vow : vowels) {

			if (i == vow) {
				vowelsCount++;
			}
		}

		for (char& con : consonants) {
			if (i == con) {
				consonantsCount++;
			}
		}
