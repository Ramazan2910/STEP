void Add(int **farr, int *farr2, int frow, int fcol, int fcounter) {
	int index = 0;
	for (int i = 0; i < frow; i++)
	{
		for (int j = 0; j < fcol; j++)
		{
			if (farr[i][j] > 77) {
				*(farr2 + index) = farr[i][j];
				index++;
			}
		}
	}

	for (int i = 0; i < fcounter; i++)
	{
		cout << *(farr2 + i) << " ";
	}
}

	delete[]farr;
}

void Delete(string* fmassiv, string fdel, int findex) {
	
	for (int i = 0; i < findex+1; i++)
	{
		if (*(fmassiv + i) == fdel) {
			int counter = i;
			while (counter < findex) {
				*(fmassiv + counter) = *(fmassiv + (counter + 1));
				counter++;
			}
		}
	}

	for (int i = 0; i < findex-1; i++)
	{
		cout << *(fmassiv + i) << endl;
	}
}

void Search(string* fmassiv, string fname, int findex) {

		}

}
