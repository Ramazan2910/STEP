void Delete(int* fmassiv, int fdelNum, int frange) {
	for (int i = 0; i < frange + 1; i++)
	{
		if (*(fmassiv + i) == fdelNum) {
			int counter = i;
			while (counter < frange) {
				*(fmassiv + counter) = *(fmassiv + (counter + 1));
				counter++;
			}
		}
	}

	for (int i = 0; i < frange - 1; i++)
	{
		cout << *(fmassiv + i) << endl;
	}
}
