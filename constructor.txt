class Phone_book {

	Phone_book(const char* name, const char* surname, const char* father_name, const char* home_number, const char* work_number, const char* own_number, const char* additional_info) {
		this->name = new char[strlen(name) + 1];
		this->surname = new char[strlen(surname) + 1];
		this->father_name = new char[strlen(father_name) + 1];

		strcpy_s(this->name, strlen(name) + 1, name);
		strcpy_s(this->surname, strlen(surname) + 1, surname);
		strcpy_s(this->father_name, strlen(father_name) + 1, father_name);
		strcpy_s(this->home_number, home_number);
		strcpy_s(this->work_number, work_number);
		strcpy_s(this->own_number, own_number);

	int Binary_search_brand(Phone_book mass[], const char* name, const char* surname, int low, int high) {


		if (high >= low) {
			int mid = low + (high - low) / 2;

			if (strcmp(name, mass[mid].name) == 0 && strcmp(surname, mass[mid].surname) == 0) {
				return mid;
			}

			if (strcmp(name, mass[mid].name) == 1 && strcmp(surname, mass[mid].surname) == 1) {
				return Binary_search_brand(mass, name, surname, mid + 1, high);
			}

			return Binary_search_brand(mass, name, surname, low, mid - 1);
		}

		return -1;
	}

class String {

	char* str;
	static int counter;

	String() {
		this->str = new char[81];
		counter++;
	}

	String(int str_size) {
		this->str = new char[str_size + 1];
		counter++;
	}

	String(const char* str) {
		this->str = new char[strlen(str) + 1];
		strcpy_s(this->str, strlen(str)+1,str);
		counter++;
	}

	~String(){
		delete[]this->str;
		counter--;
	}

	void Input() {
		cout << "Vvedite stroku: ";
		cin.getline(this->str,strlen(this->str));
	}

	void Output() {
		cout << this->str<<endl;
	}

	static int Counter() {
		return counter;
	}
	
	void Show_counter() {
		cout << this->counter << endl;
	}
};
