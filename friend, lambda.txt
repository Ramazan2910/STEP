// Task 1

class Student {
private:
	char* name;
	int grade;
public:
	Student(const char* name, int grade) {
		this->name = new char[strlen(name) + 1];
		
		strcpy_s(this->name, strlen(name) + 1, name);
		this->grade = grade;
	}

	~Student() {
		delete[]name;
	}

	void Info() {
		cout << "Student: " << this->name<< ";" << " Grade: " << this->grade << endl;

	}

	int GetGrade() {
		return this->grade;
	}

	friend void Increase(Student& student);
};

void Increase(Student& student) {
	if (student.grade < 96) {
		student.grade += 5;
		cout << "Ball uvelicen!" << endl;
	}
	else {
		cout << "Ne vozmojno uvelicet ball!"<<endl;
	}
}

class BankAccount {
private:
	char* name;
	char* surname;
	char* type_of_account;
	float balance;
public:
	BankAccount(const char* name, const char* surname, const char* type_of_account, float balance) {
		this->name = new char[strlen(name) + 1];
		this->surname = new char[strlen(surname) + 1];
		this->type_of_account = new char[strlen(type_of_account) + 1];

		strcpy_s(this->name, strlen(name) + 1, name);
		strcpy_s(this->surname, strlen(surname) + 1, surname);
		strcpy_s(this->type_of_account, strlen(type_of_account) + 1, type_of_account);

		this->balance = balance;
	}

	~BankAccount() {
		delete[]name;
		delete[]surname;
		delete[]type_of_account;
	}

	void Info() {
		cout << "Name: " << this->name << ";" << " Surname: " << this->surname << ";" << " Account Type: " << this->type_of_account << ";" << " Balance: " << this->balance << endl;
	}

	float GetBalance() {
		return this->balance;
	}

	friend void Transfer(BankAccount& transfer_acc, BankAccount& recieve_acc, float amount);
};

void Transfer(BankAccount& transfer_acc, BankAccount& recieve_acc, float amount) {
	if (strcmp(transfer_acc.type_of_account,"Credit") == 0){
		transfer_acc.balance -= amount;
		recieve_acc.balance += amount;
		cout << "Perevod prosol uspesno!" << endl;
	}
	else {
		if (transfer_acc.balance >= amount) {
			transfer_acc.balance -= amount;
			recieve_acc.balance += amount;
			cout << "Perevod prosol uspesno!" << endl;
		}
		else {
			cout << "Na scetu ne dostatocno sredstv!" << endl;
		}
	}
}
