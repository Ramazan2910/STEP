Структуры в C++ - это специальный тип данных, который позволяет объединять различные типы данных под одним именем. Они являются удобным способом организации данных в программе. Вот пример использования структур в C++:

#include <iostream>
#include <string>

// Определение структуры для представления информации о студенте
struct Student {
    std::string name;
    int age;
    float gpa;
};

int main() {
    // Создание объекта структуры Student
    Student student1;

    // Инициализация полей объекта
    student1.name = "Иван";
    student1.age = 20;
    student1.gpa = 4.5;

    // Вывод информации о студенте
    std::cout << "Имя: " << student1.name << std::endl;
    std::cout << "Возраст: " << student1.age << std::endl;
    std::cout << "Средний балл: " << student1.gpa << std::endl;

    return 0;
}

#include <iostream>
#include <string>

// Определение структуры для представления информации о книге
struct Book {
    std::string title;
    std::string author;
    int year;
    float price;
};

int main() {
    // Создание объекта структуры Book
    Book book1;

    // Инициализация полей объекта
    book1.title = "Преступление и наказание";
    book1.author = "Федор Достоевский";
    book1.year = 1866;
    book1.price = 500.0;

    // Вывод информации о книге
    std::cout << "Название: " << book1.title << std::endl;
    std::cout << "Автор: " << book1.author << std::endl;
    std::cout << "Год издания: " << book1.year << std::endl;
    std::cout << "Цена: " << book1.price << " руб." << std::endl;

    return 0;
}

#include <iostream>
#include <cstring>
#include <time.h>
#include<string>
#include <windows.h>
using namespace std;

struct date
{
    int day;
    int month;
    int year;
    char mon_name[12];
};


int main() {
    
    
    date d = { 2,5,1976,"July" };
    // Указатель p указывает на структуру типа date
    date* p = nullptr;
    // Показ содержимого структуры на экран
    // (обращение через объект)
    cout << d.day << " ";
    cout << d.month << " ";
    cout << d.year << " ";
    cout << d.mon_name << "\n\n";
    // запись адреса объекта структуры в указатель
    p = &d;
    // Показ содержимого структуры на экран
    // (обращение через указатель)
    // -> состоит из минуса - и знака больше >
    cout << p->day << " ";
    cout << p->month << " ";
    cout << p->year << " ";
    cout << p->mon_name << "\n\n";

    return 0;
}
