template<typename T>
class SinglyLinkedList {
private:
    Node<T>* head;

public:
    SinglyLinkedList() : head(nullptr) {}

    ~SinglyLinkedList() {
        clear();
    }

    // Добавление элемента в конец списка
    void append(const T& data) {
        if (!head) {
            head = new Node<T>(data);
        } else {
            Node<T>* current = head;
            while (current->next) {
                current = current->next;
            }
            current->next = new Node<T>(data);
        }
    }

    // Обход списка и вывод элементов
    void traverse() const {
        Node<T>* current = head;
        while (current) {
            std::cout << current->data << " ";
            current = current->next;
        }
        std::cout << std::endl;
    }

    // Удаление всех элементов из списка
    void clear() {
        while (head) {
            Node<T>* temp = head;
            head = head->next;
            delete temp;
        }
    }

    // Поиск элемента в списке
    bool contains(const T& data) const {
        Node<T>* current = head;
        while (current) {
            if (current->data == data) {
                return true;
            }
            current = current->next;
        }
        return false;
    }

    // Вставка элемента после указанного узла
    void insertAfter(Node<T>* prevNode, const T& data) {
        if (!prevNode) {
            std::cerr << "Previous node cannot be null" << std::endl;
            return;
        }
        Node<T>* newNode = new Node<T>(data);
        newNode->next = prevNode->next;
        prevNode->next = newNode;
    }

    // Удаление первого узла с указанными данными
    void remove(const T& data) {
        Node<T>* current = head;
        Node<T>* prev = nullptr;
        while (current) {
            if (current->data == data) {
                if (prev) {
                    prev->next = current->next;
                } else {
                    head = current->next;
                }
                delete current;
                return;
            }
            prev = current;
            current = current->next;
        }
    }

    // Замена значения узла с указанными данными на новое значение
    bool replace(const T& oldData, const T& newData) {
        Node<T>* current = head;
        while (current) {
            if (current->data == oldData) {
                current->data = newData;
                return true;
            }
            current = current->next;
        }
        return false;
    }

    // Поиск узла с указанными данными
    Node<T>* find(const T& data) const {
        Node<T>* current = head;
        while (current) {
            if (current->data == data) {
                return current;
            }
            current = current->next;
        }
        return nullptr;
    }
};

int main() {
    SinglyLinkedList<int> myList;

    myList.append(10);
    myList.append(20);
    myList.append(30);
    myList.append(40);

    std::cout << "List elements: ";
    myList.traverse();

    std::cout << "Is 20 in the list? " << (myList.contains(20) ? "Yes" : "No") << std::endl;

    myList.insertAfter(myList.find(20), 25);

    std::cout << "List elements after insertion: ";
    myList.traverse();

    myList.remove(30);

    std::cout << "List elements after removal: ";
    myList.traverse();

    myList.replace(40, 45);

    std::cout << "List elements after replacement: ";
    myList.traverse();

    return 0;
}

// Шаблон класса для узла списка
template<typename T>
class Node {
public:
    T data;      // Данные узла
    Node* prev;  // Указатель на предыдущий узел
    Node* next;  // Указатель на следующий узел

    // Конструктор для узла, принимающий данные
    Node(const T& newData) : data(newData), prev(nullptr), next(nullptr) {}
};

// Шаблон класса для двусвязного списка
template<typename T>
class DoublyLinkedList {
private:
    Node<T>* head;  // Указатель на начало списка
    Node<T>* tail;  // Указатель на конец списка

public:
    // Конструктор, инициализирующий начало и конец списка как nullptr
    DoublyLinkedList() : head(nullptr), tail(nullptr) {}

    // Деструктор, освобождающий память, занимаемую списком
    ~DoublyLinkedList() {
        clear();
    }

    // Метод добавления элемента в конец списка
    void append(const T& data) {
        Node<T>* newNode = new Node<T>(data);  // Создаем новый узел
        if (!head) {  // Если список пустой
            head = tail = newNode;  // Устанавливаем начало и конец на новый узел
        } else {  // Если список не пустой
            tail->next = newNode;  // Устанавливаем указатель на следующий узел текущего последнего узла на новый узел
            newNode->prev = tail;   // Устанавливаем указатель на предыдущий узел нового узла на текущий последний узел
            tail = newNode;         // Обновляем указатель на конец списка
        }
    }

    // Метод обхода списка в прямом направлении и вывода элементов
    void traverseForward() const {
        Node<T>* current = head;  // Указатель на текущий узел, начинаем с начала списка
        while (current) {  // Пока не достигнут конец списка
            std::cout << current->data << " ";  // Выводим данные текущего узла
            current = current->next;  // Переходим к следующему узлу
        }
        std::cout << std::endl;
    }

    // Метод обхода списка в обратном направлении и вывода элементов
    void traverseBackward() const {
        Node<T>* current = tail;  // Указатель на текущий узел, начинаем с конца списка
        while (current) {  // Пока не достигнут начало списка
            std::cout << current->data << " ";  // Выводим данные текущего узла
            current = current->prev;  // Переходим к предыдущему узлу
        }
        std::cout << std::endl;
    }

    // Метод удаления всех элементов списка и освобождения памяти
    void clear() {
        while (head) {  // Пока есть элементы в списке
            Node<T>* temp = head;  // Сохраняем указатель на текущий первый узел
            head = head->next;     // Перемещаем начало списка на следующий узел
            delete temp;           // Удаляем сохраненный узел
        }
        tail = nullptr;  // Обнуляем указатель на конец списка
    }

    // Метод проверки наличия элемента в списке
    bool contains(const T& data) const {
        Node<T>* current = head;  // Указатель на текущий узел, начинаем с начала списка
        while (current) {  // Пока не достигнут конец списка
            if (current->data == data) {  // Если данные текущего узла равны искомым данным
                return true;  // Возвращаем true, что элемент найден
            }
            current = current->next;  // Переходим к следующему узлу
        }
        return false;  // Если элемент не найден, возвращаем false
    }

    // Метод удаления элемента из списка
    bool remove(const T& data) {
        Node<T>* current = head;  // Указатель на текущий узел, начинаем с начала списка
        while (current) {  // Пока не достигнут конец списка
            if (current->data == data) {  // Если данные текущего узла равны искомым данным
                if (current == head) {  // Если удаляемый узел находится в начале списка
                    head = head->next;  // Перемещаем начало списка на следующий узел
                    if (head) {  // Если список не пустой после удаления
                        head->prev = nullptr;  // Устанавливаем предыдущий узел для нового первого узла как nullptr
                    } else {  // Если список пустой после удаления
                        tail = nullptr;  // Обнуляем указатель на конец списка
                    }
                } else if (current == tail) {  // Если удаляемый узел находится в конце списка
                    tail = tail->prev;  // Перемещаем конец списка на предыдущий узел
                    tail->next = nullptr;  // Устанавливаем следующий узел для нового последнего узла как nullptr
                } else {  // Если удаляемый узел находится в середине списка
                    current->prev->next = current->next;  // Устанавливаем указатель на следующий узел для предыдущего узла текущего узла на следующий узел текущего узла
                    current->next->prev = current->prev;  // Устанавливаем указатель на предыдущий узел для следующего узла текущего узла на предыдущий узел текущего узла
                }
                delete current;  // Удаляем узел из памяти
                return true;  // Возвращаем true, что элемент удален
            }
            current = current->next;  // Переходим к следующему узлу
        }
        return false;  // Если элемент не найден, возвращаем false
    }

    // Метод замены значения узла с указанными данными на новое значение
    bool replace(const T& oldData, const T& newData) {
        Node<T>* current = head;  // Указатель на текущий узел, начинаем с начала списка
        while (current) {  // Пока не достигнут конец списка
            if (current->data == oldData) {  // Если данные текущего узла равны искомым данным
                current->data = newData;  // Заменяем данные текущего узла на новые данные
                return true;  // Возвращаем true, что элемент заменен
            }
            current = current->next;  // Переходим к следующему узлу
        }
        return false;  // Если элемент не найден, возвращаем false
    }
};

int main() {
