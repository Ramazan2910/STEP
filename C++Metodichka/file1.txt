using namespace std;

struct Person
{
	char* name;
	char* surname;
	int age;
};

void WriteFile(const char* filename, Person* person) {
	FILE* file = nullptr;
	fopen_s(&file, filename, "a");

	if (file != nullptr) {
		fputs(person->name, file);
		fputs(" ", file);
		fputs(person->surname, file);
		fputs(" ", file);
		char* bfr = new char[4];
		_itoa_s(person->age, bfr, 4, 10);
		fputs(bfr, file);
		fputs("\n", file);

		fclose(file);
	}
}

Person* ReadPersonFromFile(const char* filename) {
	FILE* f = nullptr;

	fopen_s(&f, filename, "r");
	if (f != nullptr) {
		char* bf = new char[100]; // Hakuna Matata 42\n
		fgets(bf, 100, f);

		Person* person = new Person;
		person->name = new char[20];
		int i = 0;
		for (i = 0; bf[i] != ' '; i++) {
			person->name[i] = bf[i];
		}
		person->name[i] = '\0';
		i++;
		person->surname = new char[20];
		int j = 0;
		for (; bf[i] != ' '	; i++, j++) {
			person->surname[j] = bf[i];
		}
		person->surname[j] = '\0';
		i++;

		person->age = 0;
		return person;
	}

	nullptr;
}
