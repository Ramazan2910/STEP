Наследование шаблонов


Наследование шаблонов в C++ - это способ создания классов-шаблонов, которые наследуют параметры шаблона от других классов-шаблонов. Это позволяет создавать шаблонные классы, которые используют общие свойства и методы других шаблонных классов.

#include <iostream>

// Базовый шаблонный класс
template<typename T>
class Base {
protected:
    T data;
public:
    Base(T d) : data(d) {}
    void display() {
        std::cout << "Base: " << data << std::endl;
    }
};

// Производный шаблонный класс
template<typename T>
class Derived : public Base<T> {
public:
    Derived(T d) : Base<T>(d) {} // Конструктор, передающий параметр шаблона базовому классу
    void show() {
        std::cout << "Derived: " << this->data << std::endl; // Обращение к полю data из базового класса
    }
};

int main() {
    // Создание объекта Derived, используя тип int
    Derived<int> derivedInt(10);
    derivedInt.display(); // Метод из базового класса
    derivedInt.show();    // Метод из производного класса

    // Создание объекта Derived, используя тип double
    Derived<double> derivedDouble(3.14);
    derivedDouble.display(); // Метод из базового класса
    derivedDouble.show();    // Метод из производного класса

    return 0;
}

#include <iostream>

// Базовый шаблонный класс
template<typename T>
class Base {
protected:
    T data;
public:
    // Конструктор по умолчанию
    Base() {
        std::cout << "Base default constructor" << std::endl;
    }
    
    // Конструктор с параметром
    Base(T d) : data(d) {
        std::cout << "Base constructor with parameter" << std::endl;
    }

    // Метод вывода данных
    void display() {
        std::cout << "Base: " << data << std::endl;
    }
};

// Производный шаблонный класс
template<typename T, typename U>
class Derived : public Base<T> {
private:
    U value;
public:
    // Конструктор по умолчанию
    Derived() {
        std::cout << "Derived default constructor" << std::endl;
    }

    // Конструктор с параметром
    Derived(T d, U v) : Base<T>(d), value(v) {
        std::cout << "Derived constructor with parameters" << std::endl;
    }

    // Метод вывода данных производного класса
    void show() {
        std::cout << "Derived: " << this->data << ", " << value << std::endl; // Обращение к данным из базового класса
    }
};

int main() {
    // Создание объекта Derived с типами данных int и double
    Derived<int, double> derivedObj(10, 3.14);
    derivedObj.display(); // Метод из базового класса
    derivedObj.show();    // Метод из производного класса

    return 0;
}

Виртуальный базовый класс (Virtual Base Class) в C++ используется для разрешения проблемы "алмазного наследования" (Diamond Inheritance), которая возникает при множественном наследовании.

Проблема алмазного наследования возникает, когда класс наследуется от двух или более классов, которые, в свою очередь, наследуют от одного и того же класса. В результате в производном классе появляются две или более копии базового класса, что приводит к неоднозначности доступа к его членам и методам.

Рассмотрим пример:

#include <iostream>

// Базовый класс Animal
class Animal {
public:
    void speak() {
        std::cout << "Animal speaks" << std::endl;
    }
};

// Производный класс Dog, наследующий от Animal
class Dog : public Animal {
public:
    void speak() {
        std::cout << "Dog barks" << std::endl;
    }
};

// Производный класс Cat, наследующий от Animal
class Cat : public Animal {
public:
    void speak() {
        std::cout << "Cat meows" << std::endl;
    }
};

// Производный класс CatDog, наследующий от Dog и Cat
class CatDog : public Dog, public Cat {
};

int main() {
    CatDog myPet;
    myPet.speak(); // Ошибка компиляции из-за неоднозначности
    return 0;
}


В этом примере CatDog наследует как от класса Dog, так и от класса Cat, которые в свою очередь наследуют от Animal. При вызове метода speak() для CatDog, компилятор не знает, к какому speak() обратиться, потому что CatDog содержит две копии Animal.

Для решения этой проблемы используется виртуальное наследование.

Виртуальное наследование позволяет создавать единственный экземпляр базового класса в дочернем классе, даже если этот базовый класс унаследован несколько раз. Виртуальный базовый класс делает так, чтобы не было дублирования данных, и разрешает неоднозначность.

Используем виртуальное наследование в примере:


#include <iostream>

// Виртуальный базовый класс Animal
class Animal {
public:
    void speak() {
        std::cout << "Animal speaks" << std::endl;
    }
};

// Производный класс Dog, виртуально наследующий от Animal
class Dog : virtual public Animal {
public:
    void speak() {
        std::cout << "Dog barks" << std::endl;
    }
};

// Производный класс Cat, виртуально наследующий от Animal
class Cat : virtual public Animal {
public:
    void speak() {
        std::cout << "Cat meows" << std::endl;
    }
};

// Производный класс CatDog, наследующий от Dog и Cat
class CatDog : public Dog, public Cat {
};

int main() {
    CatDog myPet;
    myPet.speak(); // Результат: Animal speaks
    return 0;
}

Теперь при вызове speak() для CatDog происходит обращение к единственному экземпляру Animal, который наследуется виртуально. Таким образом, проблема неоднозначности решается, и компилятор знает, какой speak() вызывать.

Шаблоны классов в C++:

Шаблоны классов — это обобщенные классы, которые позволяют использовать один и тот же код для работы с разными типами данных. Они позволяют писать классы, которые будут работать с любыми типами данных (например, целые числа, числа с плавающей запятой, пользовательские типы и т.д.).

template <typename T>
class MyVector {
private:
    T* data;
    int size;

public:
    MyVector(int size) {
        this->size = size;
        data = new T[size];
    }

    T& operator[](int index) {
        return data[index];
    }

    ~MyVector() {
        delete[] data;
    }
};


Использование std::initializer_list
Класс std::initializer_list позволяет инициализировать контейнеры фиксированным набором элементов. Например, его часто используют в конструкторах:

#include <initializer_list>
#include <iostream>

class MyVector {
public:
    MyVector(std::initializer_list<int> list) {
        for (auto value : list) {
            std::cout << value << " ";
        }
    }
};

int main() {
    MyVector v = {1, 2, 3, 4};  // инициализация с помощью initializer_list
}
