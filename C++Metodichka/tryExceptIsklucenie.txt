#include <iostream>
#include <stdexcept>

using namespace std;

int main() {
    int a = 10, b = 0;
    try {
        if (b == 0) {
            throw runtime_error("Деление на ноль!"); // Генерация исключения
        }
        int result = a / b;
        cout << "Результат: " << result << endl;
    }
    catch (const runtime_error &e) {
        cout << "Произошло исключение: " << e.what() << endl;
    }
    return 0;
}

#include <iostream>

class Array {
private:
    int* data;
    int size;

public:
    Array(int size) : size(size) {
        if (size <= 0) {
            throw std::invalid_argument("Размер массива должен быть положительным.");
        }
        data = new int[size];
    }

    ~Array() {
        delete[] data;
    }

    int& operator[](int index) {
        if (index < 0 || index >= size) {
            throw std::out_of_range("Индекс вне диапазона массива.");
        }
        return data[index];
    }
};

int main() {
    try {
        // Создание массива размером 5
        Array arr(5);

        // Устанавливаем значения элементов
        for (int i = 0; i < 5; ++i) {
            arr[i] = i * 2;
        }

        // Пытаемся получить доступ к элементу с индексом за пределами массива
        std::cout << "Значение элемента с индексом 10: " << arr[10] << std::endl;
    }
    catch (const std::invalid_argument& e) {
        std::cout << "Ошибка: " << e.what() << std::endl;
    }
    catch (const std::out_of_range& e) {
        std::cout << "Ошибка: " << e.what() << std::endl;
    }
    catch (...) {
        std::cout << "Поймано неизвестное исключение." << std::endl;
    }

    return 0;
}

#include <iostream>

// Класс исключения
class MyException {
public:
    MyException(const std::string& message) : message_(message) {}

    // Метод для получения сообщения об ошибке
    const std::string& getMessage() const {
        return message_;
    }

private:
    std::string message_;
};

int main() {
    try {
        // Попытка выполнить код, который может сгенерировать исключение
        int numerator = 10;
        int denominator = 0;

        if (denominator == 0) {
            // Генерируем исключение, если делитель равен нулю
            throw MyException("Деление на ноль!");
        }

        // Результат деления
        int result = numerator / denominator;
        std::cout << "Результат: " << result << std::endl;
    }
    catch (const MyException& e) {
        // Обработка исключения
        std::cout << "Поймано исключение: " << e.getMessage() << std::endl;
    }
    catch (...) {
        // Обработка неизвестных исключений
        std::cout << "Поймано неизвестное исключение." << std::endl;
    }

    return 0;
}

#include <iostream>

class CustomException {
public:
    CustomException(const std::string& message) : message_(message) {}

    const std::string& getMessage() const {
        return message_;
    }

private:
    std::string message_;
};

int main() {
    try {
        // Simulating an exception
        throw CustomException("An error occurred!");

        // This line will not be executed if an exception is thrown
        std::cout << "This line will not be executed.\n";
    }
    catch (const CustomException& e) {
        std::cout << "Caught exception: " << e.getMessage() << std::endl;
    }
    catch (...) {
        std::cout << "Caught unknown exception.\n";
    }

    return 0;
}
