#include <iostream>
#include <string>

// Определение структуры Person
struct Person {
private:
    // Приватные поля
    std::string name;
    int age;

public:
    // Конструктор с параметрами для инициализации полей
    Person(const std::string& newName, int newAge)
        : name(newName), age(newAge) {}

    // Публичный метод для вывода информации о персоне
    void printInfo() {
        std::cout << "Name: " << name << ", Age: " << age << std::endl;
    }
};

int main() {
    // Создание экземпляра структуры с использованием конструктора
    Person person("Alice", 30);

    // Вывод информации с использованием публичного метода
    person.printInfo();

    return 0;
}
