#include <iostream>

int main() {
    // Размеры массива
    int rows = 2;
    int cols = 3;
    int depth = 4;

    // Выделение памяти для трехмерного массива
    int*** matrix = new int**[rows];
    for (int i = 0; i < rows; ++i) {
        matrix[i] = new int*[cols];
        for (int j = 0; j < cols; ++j) {
            matrix[i][j] = new int[depth];
            // Заполнение массива
            for (int k = 0; k < depth; ++k) {
                matrix[i][j][k] = i * 100 + j * 10 + k; // Примерное заполнение данными
            }
        }
    }

    // Вывод данных из массива
    for (int i = 0; i < rows; ++i) {
        for (int j = 0; j < cols; ++j) {
            for (int k = 0; k < depth; ++k) {
                std::cout << "matrix[" << i << "][" << j << "][" << k << "] = " << matrix[i][j][k] << std::endl;
            }
        }
    }

    // Освобождение памяти
    for (int i = 0; i < rows; ++i) {
        for (int j = 0; j < cols; ++j) {
            delete[] matrix[i][j];
        }
        delete[] matrix[i];
    }
    delete[] matrix;

    return 0;
}
