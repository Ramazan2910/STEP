#include <iostream>

using namespace std;

struct Person
{
	char* name;
	char* surname;
	int age;
};

void WriteFile(const char* filename, Person* person) {
	FILE* file = nullptr;
	fopen_s(&file, filename, "a");

	if (file != nullptr) {
		fputs(person->name, file);
		fputs(" ", file);
		fputs(person->surname, file);
		fputs(" ", file);
		char* bfr = new char[4];
		_itoa_s(person->age, bfr, 4, 10);
		fputs(bfr, file);
		fputs("\n", file);

		fclose(file);
	}
}

void appendPerson(Person**& people, int& size, const char* name, const char* surname, const int age) {
	Person* person = new Person;
	person->name = new char[20];
	person->surname = new char[20];

	strcpy_s(person->name, strlen(name) + 1, name);
	strcpy_s(person->surname, strlen(surname) + 1, surname);
	person->age = age;

	Person** temp = new Person * [size + 1];
	if (person != nullptr) {
		for (size_t i = 0; i < size; i++)
		{
			temp[i] = people[i];
		}
	}
	temp[size] = person;
	if (people != nullptr) {
		delete[] people;
	}
	people = temp;
	size++;
}

void ReadPersonFromFile(const char* filename, Person**& arr, int& size) {
	FILE* f = nullptr;

	fopen_s(&f, filename, "r");
	if (f != nullptr) {

		char* bf = new char[100]; // Hakuna#Matata#42\n
		while (fgets(bf, 100, f) != nullptr) {

			char* name = new char[20];
			int i = 0;
			for (i = 0; bf[i] != '#'; i++) {
				name[i] = bf[i];
			}
			name[i] = '\0';
			i++;

			char* surname = new char[20];
			int j = 0;
			for (; bf[i] != '#'; i++, j++) {
				surname[j] = bf[i];
			}
			surname[j] = '\0'; 
			i++; 
			
			char* ageBf = new char[4];
			for (j = 0; bf[i] != '\n'; i++, j++) {
				ageBf[j] = bf[i];
			}
			ageBf[j] = '\0';
			
			int age = atoi(ageBf);

			appendPerson(arr, size, name, surname, age);

			delete[] name;
			delete[] surname;
			delete[] ageBf;
		}
	}

}
