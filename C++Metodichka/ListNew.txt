#include <iostream>
#include <string>

using namespace std;

// Класс Person представляет человека с именем и возрастом
class Person {
public:
    string name;  // Имя человека
    int age;      // Возраст человека

    // Конструктор для создания объекта Person с именем и возрастом
    Person(string name, int age) : name(name), age(age) {}
};

// Структура Node представляет узел в односвязном списке
struct Node {
    Person person;  // Данные о человеке
    Node* next;     // Указатель на следующий узел

    // Конструктор для создания узла с объектом Person и указателем на следующий узел
    Node(Person p) : person(p), next(nullptr) {}
};

// Класс PersonList представляет односвязный список персон
class PersonList {
private:
    Node* head;  // Указатель на первый узел списка (начало списка)

public:
    // Конструктор инициализирует head как nullptr, так как список изначально пуст
    PersonList() : head(nullptr) {}

    // Метод для добавления человека в начало списка
    void addPersonFront(const Person& p) {
        Node* newNode = new Node(p); // Создаем новый узел
        newNode->next = head;        // Новый узел указывает на текущую голову
        head = newNode;              // Новый узел становится новой головой списка
    }

    // Метод для вывода всех персон в списке
    void printList() {
        Node* temp = head;           // Временный указатель для обхода списка
        while (temp != nullptr) {    // Пока не достигли конца списка
            cout << "Name: " << temp->person.name << ", Age: " << temp->person.age << " -> ";
            temp = temp->next;       // Переходим к следующему узлу
        }
        cout << "nullptr" << endl;   // Указываем конец списка
    }

    // Метод для удаления первого узла (персоны) из списка
    void removePersonFront() {
        if (head == nullptr) return; // Если список пуст, выходим
        Node* temp = head;           // Временный указатель для удаления головы
        head = head->next;           // Перемещаем голову на следующий узел
        delete temp;                 // Удаляем старую голову
    }

    // Деструктор освобождает память, удаляя каждый узел списка
    ~PersonList() {
        while (head != nullptr) {    // Пока список не пуст
            Node* temp = head;
            head = head->next;
            delete temp;
        }
    }
};

// Точка входа в программу
int main() {
    PersonList list;  // Создаем пустой список персон

    // Добавляем несколько персон в начало списка
    list.addPersonFront(Person("Alice", 30));
    list.addPersonFront(Person("Bob", 25));
    list.addPersonFront(Person("Charlie", 22));

    list.printList();  // Выводим список: Name: Charlie, Age: 22 -> Name: Bob, Age: 25 -> Name: Alice, Age: 30 -> nullptr

    list.removePersonFront(); // Удаляем первую персону (Charlie)
    list.printList();         // Выводим список после удаления: Name: Bob, Age: 25 -> Name: Alice, Age: 30 -> nullptr

    return 0;
}



Этот метод addPersonFront добавляет нового человека (объект Person) в начало односвязного списка. Давайте разберем каждую строку кода и объясним, что происходит на каждом шаге.

    newNode->next = head;        // Новый узел указывает на текущую голову

Пошаговое объяснение
Node* newNode = new Node(p);

Здесь создается новый узел (Node), который будет хранить данные о новом человеке (Person).
newNode — это указатель на новый объект Node. Он указывает на тот участок памяти, который выделен для нового узла.
new Node(p) вызывает конструктор Node, передавая p (объект Person) в качестве аргумента. Конструктор инициализирует поле person новым значением p и делает поле next равным nullptr.
newNode->next = head;

Эта строка связывает новый узел с остальной частью списка.
Поле next нового узла (newNode) указывает на текущий первый элемент списка (head). Таким образом, новый узел "встает перед" всеми существующими узлами, а указатель next соединяет его с прежней головой списка.
head = newNode;

На этом шаге перемещаем указатель head на новый узел.
Теперь head указывает на newNode, который становится первым элементом списка.
После этой операции новый узел официально становится головой (первым элементом) списка, а весь список "сдвигается" на один элемент вниз.

#include <iostream>

struct Node {
    int data;          // Данные узла
    Node* next;        // Указатель на следующий узел

    Node(int value) : data(value), next(nullptr) {}
};

class SinglyLinkedList {
private:
    Node* head;         // Указатель на первый элемент списка

public:
    SinglyLinkedList() : head(nullptr) {}

    // Добавление элемента в начало списка
    void push_front(int value) {
        Node* newNode = new Node(value);
        newNode->next = head;
        head = newNode;
    }

    // Добавление элемента в конец списка
    void push_back(int value) {
        Node* newNode = new Node(value);
        if (head == nullptr) {
            head = newNode;
            return;
        }
        Node* temp = head;
        while (temp->next != nullptr) {
            temp = temp->next;
        }
        temp->next = newNode;
    }

    // Удаление элемента из начала списка
    void pop_front() {
        if (head == nullptr) {
            std::cout << "List is empty\n";
            return;
        }
        Node* temp = head;
        head = head->next;
        delete temp;
    }

    // Удаление элемента из конца списка
    void pop_back() {
        if (head == nullptr) {
            std::cout << "List is empty\n";
            return;
        }
        if (head->next == nullptr) {
            delete head;
            head = nullptr;
            return;
        }
        Node* temp = head;
        while (temp->next->next != nullptr) {
            temp = temp->next;
        }
        delete temp->next;
        temp->next = nullptr;
    }

    // Поиск и удаление узла по значению
    void remove(int value) {
        if (head == nullptr) return;
        if (head->data == value) {
            pop_front();
            return;
        }
        Node* temp = head;
        while (temp->next != nullptr && temp->next->data != value) {
            temp = temp->next;
        }
        if (temp->next == nullptr) {
            std::cout << "Value not found in the list\n";
            return;
        }
        Node* toDelete = temp->next;
        temp->next = temp->next->next;
        delete toDelete;
    }

    // Вывод списка
    void print() {
        Node* temp = head;
        while (temp != nullptr) {
            std::cout << temp->data << " -> ";
            temp = temp->next;
        }
        std::cout << "nullptr\n";
    }

    // Деструктор для освобождения памяти
    ~SinglyLinkedList() {
        while (head != nullptr) {
            pop_front();
        }
    }
};

int main() {
    SinglyLinkedList list;

    list.push_back(10);
    list.push_back(20);
    list.push_front(5);
    list.print();  // Output: 5 -> 10 -> 20 -> nullptr

    list.pop_front();
    list.print();  // Output: 10 -> 20 -> nullptr

    list.push_back(30);
    list.push_back(40);
    list.remove(20);
    list.print();  // Output: 10 -> 30 -> 40 -> nullptr

    list.pop_back();
    list.print();  // Output: 10 -> 30 -> nullptr

    return 0;
}


Объяснение кода:
push_front: добавляет новый узел в начало списка.
push_back: добавляет новый узел в конец списка, используя проход по списку до последнего элемента.
pop_front: удаляет первый узел списка.
pop_back: удаляет последний узел списка.
remove: находит и удаляет первый узел с заданным значением.
print: выводит все элементы списка для визуализации.
Деструктор: удаляет все узлы при уничтожении списка, чтобы избежать утечек памяти.


Функция push_front(int value) добавляет новый элемент в начало односвязного списка. Вот как это работает, шаг за шагом:

Создаем новый узел:


Node* newNode = new Node(value);
Здесь создается новый узел (Node) с данным значением value. newNode — это указатель, который указывает на этот новый узел. Сейчас newNode создан, но еще не связан с самим списком.

Связываем новый узел со списком:

newNode->next = head;
Теперь мы устанавливаем next указатель нового узла так, чтобы он указывал на текущий первый элемент списка (head). Таким образом, новый узел "смотрит" на старую голову списка, занимая место перед ним.

Переносим голову списка на новый узел:


head = newNode;
Теперь мы устанавливаем указатель head так, чтобы он указывал на наш новый узел. Это делает новый узел первым элементом списка.

Резюмируем:
Этот метод берет новое значение, создает с ним узел, "вставляет" его в начало списка и перемещает голову списка на этот новый узел.


В данном случае head — это указатель, который указывает на первый элемент (узел) односвязного списка. Его роль заключается в том, чтобы хранить начало списка, благодаря чему мы можем добраться до всех остальных узлов, так как каждый узел содержит указатель на следующий элемент списка.

Как работает head:
Начало списка: Когда список пуст, head указывает на nullptr, то есть на пустое значение.
Добавление узлов: При добавлении узлов в список, head всегда указывает на первый узел, что позволяет получить доступ ко всему списку.
Удаление узлов: При удалении первого узла мы обновляем head, чтобы он указывал на следующий элемент. Если все элементы удалены, head снова становится nullptr.

В функции push_front, head перемещается на новый узел, так как добавляем новый элемент в начало.

Что такое head
head — это указатель, который:

На старте указывает на первый элемент списка, если он есть, или на nullptr, если список пуст.
Обозначает начало списка. Когда мы говорим о "начале списка", мы имеем в виду, что head хранит адрес (или местоположение) первого узла. И через этот узел мы можем найти все остальные узлы в списке, следуя их указателям.
Почему важен head
head — ключ к доступу ко всему списку. Мы можем представить себе head как "пульт управления": он позволяет нам знать, где начинается список, а затем, с помощью указателей в каждом узле, переходить от одного элемента к другому.

Как head работает в push_front
Теперь посмотрим, как head работает в функции push_front для добавления нового элемента в начало списка.

Создаем новый узел

Node* newNode = new Node(value);


Здесь newNode — это указатель на новый узел, в который мы добавили value. Новый узел пока еще не связан со списком.

Связываем новый узел с head

newNode->next = head;

Здесь мы говорим новому узлу, чтобы он указывал на то, на что раньше указывал head. Таким образом, newNode "подвешивается" перед текущим первым элементом списка (если такой элемент уже есть).

Переназначаем head

head = newNode;


Теперь head указывает на newNode. Таким образом, новый узел становится первым элементом списка, а прежний первый элемент теперь "следует" за ним.

Пример на основе нескольких добавлений
Представим последовательные добавления с помощью push_front(10), push_front(20), и push_front(30):

1-й вызов push_front(10):

Создается новый узел с value = 10.
Этот узел указывает на nullptr, так как список изначально пуст (head = nullptr).
Теперь head указывает на этот новый узел.

Пример на основе нескольких добавлений
Представим последовательные добавления с помощью push_front(10), push_front(20), и push_front(30):

1-й вызов push_front(10):

Создается новый узел с value = 10.
Этот узел указывает на nullptr, так как список изначально пуст (head = nullptr).
Теперь head указывает на этот новый узел.
Список:
head -> [10] -> nullptr


2-й вызов push_front(20):

Создается новый узел с value = 20.
Этот узел указывает на старый head (то есть на узел [10]).
Теперь head переназначен на новый узел, и он стал первым элементом.
Список:

head -> [20] -> [10] -> nullptr

3-й вызов push_front(30):

Создается новый узел с value = 30.
Этот узел указывает на текущий head (то есть на узел [20]).
Теперь head переназначен на новый узел, и он стал первым элементом.
Список:

head -> [30] -> [20] -> [10] -> nullptr


Теперь каждый раз, когда мы добавляем новый элемент в начало с помощью push_front, head обновляется, чтобы всегда указывать на самый "свежий" первый элемент списка.
