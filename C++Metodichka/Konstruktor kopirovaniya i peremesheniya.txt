#include <iostream>
#include <cstring>
using namespace std;

class Student {
private:
    char* name;
    int age;
    float cpp_grade; // Оценка по C++

public:
    // Конструктор по умолчанию
    Student() {
        name = nullptr;
        age = 0;
        cpp_grade = 0.0;
    }

    // Конструктор с параметрами
    Student(const char* studentName, int studentAge, float grade) {
        int length = strlen(studentName);
        name = new char[length + 1]; // Выделение памяти для имени
        strcpy(name, studentName); // Копирование имени
        age = studentAge;
        cpp_grade = grade; // Успеваемость по C++
    }

    // Конструктор копирования
    Student(const Student& other) {
        cout << "Конструктор копирования сработал!" << endl;

        // Копируем имя, выделив под него новую память
        int length = strlen(other.name);
        name = new char[length + 1];
        strcpy(name, other.name);

        // Копируем возраст и оценку по C++
        age = other.age;
        cpp_grade = other.cpp_grade;
    }

    // Оператор присваивания
    Student& operator=(const Student& other) {
        if (this == &other) {
            return *this; // Защита от самоприсваивания
        }

        // Освобождаем старую память
        if (name != nullptr) {
            delete[] name;
        }

        // Копируем имя
        int length = strlen(other.name);
        name = new char[length + 1];
        strcpy(name, other.name);

        // Копируем возраст и оценку по C++
        age = other.age;
        cpp_grade = other.cpp_grade;

        return *this;
    }

    // Метод для вывода информации о студенте
    void printInfo() const {
        cout << "Имя студента: " << name << endl;
        cout << "Возраст: " << age << endl;
        cout << "Оценка по C++: " << cpp_grade << endl;
    }

    // Деструктор для освобождения динамически выделенной памяти
    ~Student() {
        delete[] name;
    }
};

int main() {
    // Создаем студента с помощью конструктора с параметрами
    Student student1("Tamerlan", 21, 85.5);
    student1.printInfo();

    // Создаем студента с помощью конструктора копирования
    Student student2 = student1; // Конструктор копирования вызывается здесь
    student2.printInfo();

    return 0;
}

#include <iostream>
#include <cstring>
using namespace std;

class Student {
private:
    char* name;
    int age;

public:
    // Конструктор по умолчанию
    Student() {
        name = nullptr;
        age = 0;
    }

    // Конструктор с параметрами
    Student(const char* name, int age) {
        int length = strlen(name);
        this->name = new char[length + 1];  // Выделяем память под имя
        strcpy_s(this->name, length + 1, name);  // Копируем имя с использованием strcpy_s
        this->age = age;                          // Копируем возраст
    }

    // Конструктор копирования
    Student(const Student& other) {
        if (other.name != nullptr) {
            int length = strlen(other.name);
            this->name = new char[length + 1];  // Выделяем память для копии имени
            strcpy_s(this->name, length + 1, other.name);  // Копируем имя с использованием strcpy_s
        }
        else {
            this->name = nullptr;
        }
        this->age = other.age;  // Копируем возраст
    }

    // Оператор присваивания
    Student& operator=(const Student& other) {
        if (this == &other) {  // Проверка на самоприсваивание
            return *this;
        }

        // Освобождаем память, если она уже была выделена
        if (this->name != nullptr) {
            delete[] this->name;
        }

        if (other.name != nullptr) {
            int length = strlen(other.name);
            this->name = new char[length + 1];  // Выделяем новую память
            strcpy_s(this->name, length + 1, other.name);  // Копируем данные с использованием strcpy_s
        }
        else {
            this->name = nullptr;
        }
        this->age = other.age;  // Копируем возраст

        return *this;
    }

    // Метод для вывода информации о студенте
    void print() const {
        cout << "Name: " << (name ? name : "No Name") << ", Age: " << age << endl;
    }

    // Деструктор
    ~Student() {
        delete[] name;  // Освобождаем память
    }
};

class StudentList {
private:
    Student* students;  // Массив студентов
    int size;           // Количество студентов

public:
    // Конструктор по умолчанию
    StudentList(int size) {
        this->size = size;
        students = new Student[size];  // Выделяем память для массива студентов
    }

    // Конструктор копирования для списка студентов
    StudentList(const StudentList& other) {
        this->size = other.size;
        students = new Student[size];  // Выделяем память для копии массива студентов
        for (int i = 0; i < size; i++) {
            students[i] = other.students[i];  // Копируем каждого студента
        }
    }

    // Оператор присваивания для списка студентов
    StudentList& operator=(const StudentList& other) {
        if (this == &other) {  // Проверка на самоприсваивание
            return *this;
        }

        // Освобождаем старую память
        delete[] students;

        this->size = other.size;
        students = new Student[size];  // Выделяем память для нового массива студентов
        for (int i = 0; i < size; i++) {
            students[i] = other.students[i];  // Копируем студентов
        }

        return *this;
    }

    // Метод для добавления студента
    void addStudent(int index, const Student& student) {
        if (index >= 0 && index < size) {
            students[index] = student;  // Используем оператор присваивания
        }
    }

    // Метод для вывода списка студентов
    void print() const {
        for (int i = 0; i < size; i++) {
            students[i].print();
        }
    }

    // Деструктор
    ~StudentList() {
        delete[] students;  // Освобождаем память для массива студентов
    }
};

int main() {
    // Создаем список студентов
    StudentList list(2);

    // Добавляем студентов в список
    Student student1("Alice", 20);
    Student student2("Bob", 22);

    list.addStudent(0, student1);
    list.addStudent(1, student2);

    // Выводим список студентов
    cout << "Original List:" << endl;
    list.print();

    // Копируем список студентов
    StudentList copiedList = list;
    cout << "Copied List:" << endl;
    copiedList.print();

    return 0;
}

#include <iostream>
#include <cstring>

using namespace std;

class Student {
private:
    char* name;
    int age;

public:
    // Конструктор по умолчанию
    Student() : name(nullptr), age(0) {}

    // Конструктор с параметрами
    Student(const char* name, int age) {
        int length = strlen(name);
        this->name = new char[length + 1];
        strcpy(this->name, name);
        this->age = age;
    }

    // Конструктор копирования
    Student(const Student& other) {
        cout << "Copy Constructor called" << endl;
        int length = strlen(other.name);
        this->name = new char[length + 1];
        strcpy(this->name, other.name);
        this->age = other.age;
    }

    // Конструктор перемещения
    Student(Student&& other) noexcept {
        cout << "Move Constructor called" << endl;
        this->name = other.name;  // Перемещаем указатель на имя
        this->age = other.age;

        other.name = nullptr;  // Очищаем указатель у временного объекта
        other.age = 0;
    }

    // Оператор присваивания перемещением
    Student& operator=(Student&& other) noexcept {
        cout << "Move Assignment Operator called" << endl;
        if (this != &other) {
            delete[] name;  // Освобождаем текущую память

            // Перемещаем ресурсы
            this->name = other.name;
            this->age = other.age;

            // Очищаем временный объект
            other.name = nullptr;
            other.age = 0;
        }
        return *this;
    }

    // Деструктор
    ~Student() {
        delete[] name;  // Освобождаем память
    }

    // Метод для вывода информации
    void print() const {
        cout << "Name: " << (name ? name : "No Name") << ", Age: " << age << endl;
    }
};

#include <iostream>

class MyClass {
public:
    int x;

    // Конструктор по умолчанию
    MyClass(int val) : x(val) {
        std::cout << "Default constructor called" << std::endl;
    }

    // Конструктор копирования
    MyClass(const MyClass& other) {
        x = other.x;
        std::cout << "Copy constructor called" << std::endl;
    }

    // Метод для вывода значения
    void print() const {
        std::cout << "x: " << x << std::endl;
    }
};

int main() {
    MyClass obj1(10);       // Вызов конструктора по умолчанию
    MyClass obj2 = obj1;    // Вызов конструктора копирования

    obj1.print();
    obj2.print();

    return 0;
}
