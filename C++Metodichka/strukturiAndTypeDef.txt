using namespace std;

struct Person {
	const char* name;
	const char* surname;
	int age;
};

struct Student {
	Person student;
	const char* homework;

	void InfoStudent() {
		cout << student.name << ' ' << student.surname << ' ' << homework<< endl;
	}
};

struct Teacher {
	Person teacher;
	int ocenkaUchitela;

	void InfoTeacher() {
		cout << teacher.name << ' ' << teacher.surname << ' ' << ocenkaUchitela << endl;
	}
};



int main()
{
	
	Student student1;
	student1.student.name = "Kerim";
	student1.student.surname = "Baliev";
	student1.homework = "Lesson21";
	student1.InfoStudent();
	
	Student student2;
	student2.student.name = "Ismail";
	student2.student.surname = "Abbasbeyli";
	student2.homework = "GPTspisalSamPriznayus";
	student2.InfoStudent();

	teacher1.teacher.name = "Tamerlan";
	teacher1.teacher.surname = "Hasanov";

	teacher1.InfoTeacher();

	
}
