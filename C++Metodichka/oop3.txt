#include<iostream>
#include<locale>
#include<string>
#include <cstdlib> 
#include <time.h>
#include <algorithm> 
using namespace std;


class MyStr {
private:
	char* str;

public:
	MyStr() {
		str = nullptr;
	}

	MyStr(const char* str) {
		int length = strlen(str);
		this->str = new char[length + 1];
		for (int i = 0; i < length; i++)
		{
			this->str[i] = str[i];
		}
		this->str[length] = '\0';
	}

	MyStr(const MyStr& other) { // const copy
		
		int length = strlen(other.str);
		this->str = new char[length + 1];
		for (int i = 0; i < length; i++)
		{
			this->str[i] = other.str[i];
		}
		this->str[length] = '\0';

	}

	MyStr& operator=(const MyStr& other) {
		if (this->str != nullptr) {
			delete[]str;
		}
		int length = strlen(other.str);
		this->str = new char[length + 1];
		for (int i = 0; i < length; i++)
		{
			this->str[i] = other.str[i];
		}
		this->str[length] = '\0';

		return *this;
	}

	MyStr operator+(const MyStr &other) {
		MyStr newStr;
		int thisLength = strlen(this->str);
		int otherLength = strlen(other.str);
		
		newStr.str = new char[thisLength + otherLength + 1];
		int i = 0;
		for (; i < thisLength; i++)
		{
			newStr.str[i] = this->str[i];
		}
		
		for (int j = 0; j < otherLength; j++,i++)
		{
			newStr.str[i] = other.str[j];
		}

		newStr.str[thisLength + otherLength] = '\0';

		return newStr;

	}

	void Print() {
		cout << str;
	}

	~MyStr(){
		delete[] this->str;
	}
};


int main() {
	MyStr str1("Tamerlan");
	MyStr str2("Has");
	MyStr res;
	res = str1 + str2;
	res.Print();
}
